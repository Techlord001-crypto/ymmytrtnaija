<!-- ================= Cart Section ================= -->
<div class="p-6 bg-gray-100 rounded-xl shadow-lg mt-6 max-w-lg mx-auto">
  <h3 class="text-2xl font-bold mb-4">🛒 Your Cart</h3>
  <ul id="cart-items" class="space-y-2"></ul>
  <p class="mt-4 text-lg font-semibold">Total: $<span id="cart-total">0</span></p>
</div>

<script>
  let cart = [];
  let total = 0;

  const buttons = document.querySelectorAll(".add-to-cart");
  const cartItems = document.getElementById("cart-items");
  const cartTotal = document.getElementById("cart-total");

  // Add to Cart
  buttons.forEach(button => {
    button.addEventListener("click", () => {
      const name = button.getAttribute("data-name");
      const price = parseFloat(button.getAttribute("data-price"));

      // Check if item already exists
      const existing = cart.find(item => item.name === name);
      if (existing) {
        existing.qty += 1;
      } else {
        cart.push({ name, price, qty: 1 });
      }

      total += price;
      updateCartUI();
    });
  });

  // Update Cart UI
  function updateCartUI() {
    cartItems.innerHTML = "";
    cart.forEach((item, index) => {
      const li = document.createElement("li");
      li.className = "flex justify-between items-center bg-white p-2 rounded-lg shadow";

      li.innerHTML = `
        <span>${item.name} (x${item.qty}) - $${(item.price * item.qty).toFixed(2)}</span>
        <button class="remove-item bg-red-500 hover:bg-red-600 text-white px-2 py-1 rounded" data-index="${index}">
          Remove
        </button>
      `;
      cartItems.appendChild(li);
    });
    cartTotal.textContent = total.toFixed(2);

    // Attach remove event to buttons
    document.querySelectorAll(".remove-item").forEach(btn => {
      btn.addEventListener("click", () => {
        const index = btn.getAttribute("data-index");
        total -= cart[index].price * cart[index].qty;
        cart.splice(index, 1);
        updateCartUI();
      });
    });
  }
</script>
